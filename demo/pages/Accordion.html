Accordion

Demo showcasing all configuration options for Accordion.

Available Options

{
	closeOthers: false, // If true, only one accordion item can be open at a time within the same group
	icon: "plus", // 'plus', 'arrow', or 'none'
	animationDuration: 500, // Matches the CSS transition duration for WAAPI fallback
}

details[data-component="Accordion"] {
  @supports (interpolate-size: allow-keywords) {
    &::details-content {
      transition: height 0.5s ease, opacity 0.5s ease, display 0.5s ease allow-discrete, content-visibility 0.5s ease allow-discrete;
      height: 0;
      opacity: 0;
      overflow: clip;
      display: block;
    }
  }

  &[open]::details-content {
    height: auto;
    opacity: 1;
  }

  .content {
    padding: 0 1rem 1rem;

    > :first-child {
      margin-top: 0;
    }

    > :last-child {
      margin-bottom: 0;
    }
  }

  summary {
    cursor: pointer;
    user-select: none;
    list-style: none;
    display: flex;
    justify-content: space-between;
    align-items: center;
    padding: 1rem;
    border-radius: 4px;
    transition: background-color 0.3s ease;

    &:hover {
      background-color: rgba(0, 0, 0, 0.05);
    }

    &::-webkit-details-marker {
      display: none;
    }
  }

  .accordion-icon {
    flex-shrink: 0;
    margin-left: 1rem;
    transition: transform 0.3s ease;

    &.accordion-icon--plus {
      .vertical-line {
        transition: transform 0.3s ease;
        transform-origin: center;
      }
    }
  }

  &[open] {
    .accordion-icon--arrow {
      transform: rotate(180deg);
    }

    .accordion-icon--plus {
      .vertical-line {
        transform: rotate(90deg);
      }
    }
  }
}
